3. 获取图片url
python scripts/3_get_image_urls.py \
  --html-counts outputs/final/htmls/2015/html_counts.json \
  --html-root outputs/final/htmls \
  --workers 128 \
  --report-prefix outputs/final/analysis/2015/image_scan

4. 扫描有效域名
python scripts/4_scan_domain.py \
  --input outputs/final/analysis/2015/image_scan_domain_urls.json \
  --output-prefix outputs/final/analysis/2015/domain_scan \
  --workers 128 \
  --timeout 5 \
  --max-prefixes 6 \
  --max-samples 3

5. 直接扫描并下载图片
python scripts/5_scan_and_download_images.py \
  --accessible-domains outputs/final/analysis/2015/domain_scan_accessible_domains.json \
  --domain-urls outputs/final/analysis/2015/image_scan_domain_urls.json \
  --output-prefix outputs/final/images/2015_test/image_scan_4_4 \
  --output-root outputs/final/images/2015_test \
  --workers 64 \
  --splits 4 \
  --index 4 \
  --resume

6. 分析访问结果，同时合并所有下载的文件
python scripts/6_analyze_image_size_scan.py \

  --output-prefix outputs/final/analysis/2015/analyze_image_size_final \
  --merge-workers 16 \
  --workers 64 \
  --chunk-size 50000 \
  --topk 50

7. 获取url的caption
python scripts/7_get_image_text_pairs_jsonl.py \
  --image-size-detailed outputs/final/images/2015_test/merged_success.jsonl \
  --image-scan-per-html outputs/final/analysis/2015/image_scan_records.jsonl \
  --output outputs/final/pairs/2015/image_text_pairs_final.jsonl \
  --workers 64 \
  --load-workers 16 \
  --load-chunk-size 50000 \
  --resume

9. 相似度统计和过滤
获取统计信息
python scripts/9_analyze_caption_similarity_jsonl.py \
  --input outputs/final/pairs/2015/image_text_pairs_new_with_caption_similarity_filtered.jsonl \
  --output-sorted outputs/final/pairs/2015/image_text_pairs_new_with_caption_similarity_sorted.jsonl \
  --output-stats outputs/final/analysis/2015/image_text_pairs_new_with_caption_similarity_stats.json \

对应选择阈值来过滤
python scripts/9_filter_caption_similarity_jsonl.py \
    --input outputs/final/pairs/2015/image_text_pairs_new_with_caption_similarity_sorted.jsonl \
